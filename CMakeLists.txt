cmake_minimum_required(VERSION 3.20)
project(M1zuki VERSION 1.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
if(WIN32)
  set(DESKTOP_SOURCES src/desktop_windows.cpp src/desktop_windows.h)
  set(PROTECTION_SOURCES src/captureprotection_windows.cpp src/captureprotection_windows.h)
  set(DESKTOP_LIBS user32 gdi32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
elseif(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  set(DESKTOP_SOURCES src/desktop_macos.mm src/desktop_macos.h)
  set(PROTECTION_SOURCES src/captureprotection_macos.mm src/captureprotection_macos.h)
  set(DESKTOP_LIBS "-framework AppKit" "-framework Carbon" "-framework ApplicationServices" "-framework ScreenCaptureKit")
  set_source_files_properties(src/desktop_macos.mm src/captureprotection_macos.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
else()
  find_package(Qt6 REQUIRED COMPONENTS DBus)
  find_package(KF6GlobalAccel REQUIRED)
  find_package(KF6Config REQUIRED)
  find_package(LayerShellQt QUIET)
  set(DESKTOP_SOURCES src/desktop.cpp src/mousebindings.cpp)
  set(PROTECTION_SOURCES src/captureprotection.cpp)
  set(DESKTOP_LIBS Qt6::DBus KF6::GlobalAccel KF6::ConfigCore)
endif()
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

add_library(answer_core src/openrouter.cpp src/settings.cpp)
target_include_directories(answer_core PUBLIC src)
target_link_libraries(answer_core PUBLIC Qt6::Gui Qt6::Network)

add_executable(mini-answer src/main.cpp src/controller.cpp ${DESKTOP_SOURCES} src/overlay.cpp ${PROTECTION_SOURCES})
set_target_properties(mini-answer PROPERTIES OUTPUT_NAME m1zuki)
qt_add_resources(mini-answer app_images
  PREFIX "/"
  BASE "assets"
  FILES assets/mini-answer-character.png assets/check.svg assets/chevron-up.svg assets/chevron-down.svg
)
target_link_libraries(mini-answer PRIVATE answer_core Qt6::Widgets ${DESKTOP_LIBS})
if(WIN32)
  set_target_properties(mini-answer PROPERTIES WIN32_EXECUTABLE TRUE)
  target_link_libraries(answer_core PUBLIC advapi32)
  target_link_libraries(mini-answer PRIVATE dwmapi)
elseif(APPLE)
  set_target_properties(mini-answer PROPERTIES MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_BUNDLE_NAME "m1zuki"
    MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.mmionya.m1zuki"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/Info.plist.in")
  install(TARGETS mini-answer BUNDLE DESTINATION .)
endif()
if(LayerShellQt_FOUND)
  target_compile_definitions(mini-answer PRIVATE HAVE_LAYER_SHELL)
  target_link_libraries(mini-answer PRIVATE LayerShellQt::Interface)
endif()


include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  foreach(test_name openrouter settings)
    add_executable(test_${test_name} tests/test_${test_name}.cpp)
    target_link_libraries(test_${test_name} PRIVATE answer_core Qt6::Test)
    add_test(NAME ${test_name} COMMAND test_${test_name})
    set_tests_properties(${test_name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endforeach()
  if(WIN32)
    add_executable(test_desktop tests/test_desktop_windows.cpp ${DESKTOP_SOURCES})
  elseif(APPLE)
    add_executable(test_desktop tests/test_desktop_macos.cpp ${DESKTOP_SOURCES})
  else()
    add_executable(test_desktop tests/test_desktop.cpp ${DESKTOP_SOURCES})
  endif()
  target_include_directories(test_desktop PRIVATE src)
  target_link_libraries(test_desktop PRIVATE Qt6::Widgets Qt6::Test ${DESKTOP_LIBS})
  add_test(NAME desktop COMMAND test_desktop)
  add_executable(test_overlay tests/test_overlay.cpp src/overlay.cpp)
  target_include_directories(test_overlay PRIVATE src)
  target_link_libraries(test_overlay PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME overlay COMMAND test_overlay)
  set_tests_properties(desktop overlay PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(WIN32)
    set_tests_properties(desktop overlay PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows")
  endif()
  if(UNIX AND NOT APPLE)
  add_executable(test_mousebindings tests/test_mousebindings.cpp src/mousebindings.cpp)
  target_include_directories(test_mousebindings PRIVATE src)
  target_link_libraries(test_mousebindings PRIVATE Qt6::Gui Qt6::Test KF6::ConfigCore)
  add_test(NAME mousebindings COMMAND test_mousebindings)
  set_tests_properties(mousebindings PROPERTIES ENVIRONMENT
    "QT_QPA_PLATFORM=offscreen;DBUS_SESSION_BUS_ADDRESS=unix:path=/tmp/m1zuki-test-no-session-bus")
  endif()
  if(LayerShellQt_FOUND)
    # Explicit, manual Wayland integration probe. Never run by ordinary CTest.
    add_executable(privacy_probe tests/privacy_probe.cpp src/captureprotection.cpp src/overlay.cpp)
    target_include_directories(privacy_probe PRIVATE src)
    target_compile_definitions(privacy_probe PRIVATE HAVE_LAYER_SHELL)
    target_link_libraries(privacy_probe PRIVATE Qt6::Widgets Qt6::DBus LayerShellQt::Interface)
  endif()
  if(WIN32)
    add_executable(test_windows_integration tests/test_windows_integration.cpp ${PROTECTION_SOURCES} src/overlay.cpp src/controller.cpp ${DESKTOP_SOURCES})
    qt_add_resources(test_windows_integration test_images PREFIX "/" BASE "assets"
      FILES assets/mini-answer-character.png assets/check.svg assets/chevron-up.svg assets/chevron-down.svg)
    target_include_directories(test_windows_integration PRIVATE src)
    target_link_libraries(test_windows_integration PRIVATE answer_core Qt6::Widgets Qt6::Test user32 gdi32 dwmapi)
    add_test(NAME windows_integration COMMAND test_windows_integration)
  endif()
endif()
if(APPLE AND BUILD_TESTING)
  add_executable(test_captureprotection_macos tests/test_captureprotection_macos.cpp ${PROTECTION_SOURCES})
  target_include_directories(test_captureprotection_macos PRIVATE src)
  target_link_libraries(test_captureprotection_macos PRIVATE Qt6::Widgets Qt6::Test ${DESKTOP_LIBS})
  add_test(NAME captureprotection_macos COMMAND test_captureprotection_macos)
  set_tests_properties(captureprotection_macos PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME captureprotection_macos_native COMMAND test_captureprotection_macos)
  set_tests_properties(captureprotection_macos_native PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=cocoa")
endif()
if(NOT APPLE)
install(TARGETS mini-answer RUNTIME DESTINATION bin)
install(FILES assets/mini-answer-character.svg DESTINATION share/icons/hicolor/scalable/apps RENAME m1zuki.svg)
# Old desktop entries and pinned launchers must resolve to the new artwork too.
install(FILES assets/mini-answer-character.svg DESTINATION share/icons/hicolor/scalable/apps RENAME mini-answer.svg)
install(FILES assets/mini-answer-character.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES assets/mini-answer-character-credits.txt DESTINATION share/doc/mini-answer)

endif()
